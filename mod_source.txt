    private static boolean trySetUniform(Minecraft mc, float value) {
        try {
            if (cachedUniform == null) {
                if (postEffectField == null) {
                    postEffectField = ObfuscationReflectionHelper.findField(GameRenderer.class, "postEffect");
                    postEffectField.setAccessible(true);
                }

                Object effectObj = postEffectField.get(mc.gameRenderer);
                if (!(effectObj instanceof PostChain postChain)) {
                    return false;
                }

                if (passesField == null) {
                    passesField = ObfuscationReflectionHelper.findField(PostChain.class, "passes");
                    passesField.setAccessible(true);
                }

                Object passesObj = passesField.get(postChain);
                if (!(passesObj instanceof List<?> passes)) {
                    return false;
                }

                for (Object passObj : passes) {
                    if (!(passObj instanceof PostPass pass)) continue;
                    EffectInstance effect = pass.getEffect();
                    if (effect == null) continue;
                    Uniform uniform = effect.getUniform("Intensity");
                    if (uniform != null) {
                        cachedUniform = uniform;
                        break;
                    }
                }

                if (cachedUniform == null) {
                    return false;
                }
            }

            cachedUniform.set(value);
            return true;
        } catch (Exception e) {
            LOGGER.warn("Smooth grayscale fade unavailable, falling back to instant toggle: {}", e.getMessage());
            return false;
        }
    }

    public void onComputeCameraAngles(ViewportEvent.ComputeCameraAngles event) {
        float supp = getSuppression();
        float fl = getFlash();
        if (supp <= 0.001f && fl <= 0.001f) return;

        long t = System.currentTimeMillis();

        // Р Р°Р·РЅС‹Рµ РїРµСЂРёРѕРґС‹ Рё "СЃРёРґС‹" РЅР° РєР°Р¶РґСѓСЋ РѕСЃСЊ/РіР°СЂРјРѕРЅРёРєСѓ - С‡С‚РѕР±С‹ РґРІРёР¶РµРЅРёРµ РЅРµ РІС‹РіР»СЏРґРµР»Рѕ
        // СЃРёРЅС…СЂРѕРЅРЅС‹Рј Рё РїСЂРµРґСЃРєР°Р·СѓРµРјС‹Рј (РєР°Рє Р±С‹Р»Рѕ Р±С‹ СЃ РѕРґРёРЅР°РєРѕРІС‹РјРё РїРѕ С„Р°Р·Рµ СЃРёРЅСѓСЃРѕРёРґР°РјРё)
        double yawShake = smoothNoise(t, 90, 1) * 6.0 + smoothNoise(t, 47, 2) * 5.5;   // Р±С‹Р»Рѕ 5.0 Рё 2.2
        double pitchShake = smoothNoise(t, 110, 3) * 3.84 + smoothNoise(t, 61, 4) * 3.5; // Р±С‹Р»Рѕ 3.2 Рё 1.4
        double rollShake = smoothNoise(t, 130, 5) * 1.92; // Р±С‹Р»Рѕ 1.6

        float suppIntensity = supp * supp; // РєРІР°РґСЂР°С‚РёС‡РЅР°СЏ РєСЂРёРІР°СЏ - СЃР»Р°Р±РѕРµ РїРѕРґР°РІР»РµРЅРёРµ РїРѕС‡С‚Рё РЅРµ С‚СЂСЏСЃС‘С‚, СЃРёР»СЊРЅРѕРµ - РѕС‰СѓС‚РёРјРѕ
        double flashSway = smoothNoise(t, 400, 6) * 2.0 * fl; // РѕС‚ РєРѕРЅС‚СѓР·РёРё - РјРµРґР»РµРЅРЅС‹Р№ "РїР»С‹РІСѓС‰РёР№" РєСЂРµРЅ

        event.setYaw((float) (event.getYaw() + yawShake * suppIntensity));
        event.setPitch((float) (event.getPitch() + pitchShake * suppIntensity));
        event.setRoll((float) (event.getRoll() + rollShake * suppIntensity + flashSway));
    }

    /** РҐСЌС€-С€СѓРј: РґРµС‚РµСЂРјРёРЅРёСЂРѕРІР°РЅРЅРѕРµ РїСЃРµРІРґРѕСЃР»СѓС‡Р°Р№РЅРѕРµ С‡РёСЃР»Рѕ РІ [-1, 1] РґР»СЏ С†РµР»РѕРіРѕ РёРЅРґРµРєСЃР°. */
    private static float hash(long index, long seed) {
        long x = index * 0x9E3779B97F4A7C15L + seed * 0xBF58476D1CE4E5B9L;
        x = (x ^ (x >>> 33)) * 0xFF51AFD7ED558CCDL;
        x = (x ^ (x >>> 33)) * 0xC4CEB9FE1A85EC53L;
        x = x ^ (x >>> 33);
        return ((x & 0xFFFFFF) / (float) 0xFFFFFF) * 2f - 1f;
    }

    /**
     * РџР»Р°РІРЅС‹Р№ РїСЃРµРІРґРѕСЃР»СѓС‡Р°Р№РЅС‹Р№ С€СѓРј РІРѕ РІСЂРµРјРµРЅРё: Р·РЅР°С‡РµРЅРёРµ РјРµРЅСЏРµС‚СЃСЏ РєР°Р¶РґС‹Рµ {@code periodMs} РјСЃ,
     * Р° РјРµР¶РґСѓ "РѕРїРѕСЂРЅС‹РјРё С‚РѕС‡РєР°РјРё" РёРЅС‚РµСЂРїРѕР»РёСЂСѓРµС‚СЃСЏ smoothstep'РѕРј (Р±РµР· РѕС‰СѓС‰РµРЅРёСЏ СЃРёРЅСѓСЃРѕРёРґС‹/РїСЂСѓР¶РёРЅС‹).
     */
    private static float smoothNoise(long timeMs, long periodMs, long seed) {
        long index = timeMs / periodMs;
        float frac = (timeMs % periodMs) / (float) periodMs;
        float a = hash(index, seed);
        float b = hash(index + 1, seed);
        float t = frac * frac * (3f - 2f * frac); // smoothstep
        return a + (b - a) * t;
    }

    private static void fillHorizontalGradientStepped(GuiGraphics graphics, int x1, int y1, int x2, int y2,
                                                        int colorLeft, int colorRight) {
        int totalWidth = x2 - x1;
        if (totalWidth <= 0) return;

        int steps = Math.min(GRADIENT_STEPS, totalWidth);
        for (int i = 0; i < steps; i++) {
            float t0 = (float) i / steps;
            float t1 = (float) (i + 1) / steps;

            int sx1 = x1 + Math.round(t0 * totalWidth);
            int sx2 = x1 + Math.round(t1 * totalWidth);
            if (sx2 <= sx1) continue;

            float tMid = (t0 + t1) / 2f;
            int color = lerpColor(colorLeft, colorRight, tMid);

            graphics.fill(sx1, y1, sx2, y2, color);
        }
    }

    private static int lerpColor(int colorFrom, int colorTo, float t) {
        int aF = (colorFrom >> 24) & 0xFF, rF = (colorFrom >> 16) & 0xFF, gF = (colorFrom >> 8) & 0xFF, bF = colorFrom & 0xFF;
        int aT = (colorTo >> 24) & 0xFF, rT = (colorTo >> 16) & 0xFF, gT = (colorTo >> 8) & 0xFF, bT = colorTo & 0xFF;

        int a = Math.round(aF + (aT - aF) * t);
        int r = Math.round(rF + (rT - rF) * t);
        int g = Math.round(gF + (gT - gF) * t);
        int b = Math.round(bF + (bT - bF) * t);

        return withAlpha((r << 16) | (g << 8) | b, a);
    }

    private static int withAlpha(int rgb, int alpha) {
        return (alpha << 24) | (rgb & 0x00FFFFFF);
    }

            // (TACZ Рё РїРѕРґРѕР±РЅС‹Рµ), РєРѕС‚РѕСЂРѕРµ РЅР°РЅРѕСЃРёС‚ СѓСЂРѕРЅ "РјРіРЅРѕРІРµРЅРЅС‹Рј Р»СѓС‡РѕРј" РѕС‚
            // СЃС‚СЂРµР»РєР° Р±РµР· СЃРїР°РІРЅР° С„РёР·РёС‡РµСЃРєРѕР№ РїСѓР»Рё. РћС‚Р»РёС‡Р°РµРј РІС‹СЃС‚СЂРµР» РѕС‚ СЂСѓРєРѕРїР°С€РЅРѕР№
            // Р°С‚Р°РєРё РїРѕ РґРёСЃС‚Р°РЅС†РёРё РјРµР¶РґСѓ Р°С‚Р°РєСѓСЋС‰РёРј Рё РёРіСЂРѕРєРѕРј.
            double distSqr = causer.distanceToSqr(player);

    @SubscribeEvent
    public void onExplosionDetonate(ExplosionEvent.Detonate event) {
        if (!(event.getLevel() instanceof ServerLevel level)) return;

        Vec3 center = event.getExplosion().getPosition();

        for (ServerPlayer player : level.players()) {
            double dist = player.position().distanceTo(center);
            if (dist > EXPLOSION_RADIUS) continue;

            // Р§РµРј Р±Р»РёР¶Рµ Рє СЌРїРёС†РµРЅС‚СЂСѓ, С‚РµРј СЃРёР»СЊРЅРµРµ РєРѕРЅС‚СѓР·РёСЏ. РЈ СЃР°РјРѕРіРѕ СЌРїРёС†РµРЅС‚СЂР° - РїРѕС‡С‚Рё РјР°РєСЃРёРјСѓРј.
            float falloff = (float) (1.0 - (dist / EXPLOSION_RADIUS));
            falloff = falloff * falloff; // РєРІР°РґСЂР°С‚РёС‡РЅРѕРµ Р·Р°С‚СѓС…Р°РЅРёРµ - СЂРµР°Р»РёСЃС‚РёС‡РЅРµРµ, С‡РµРј Р»РёРЅРµР№РЅРѕРµ

            float flashIntensity = Math.min(1.0f, falloff * 1.2f);
            float suppressionKick = Math.min(1.0f, falloff * 0.9f);

            if (flashIntensity <= 0.02f) continue;

            NetworkHandler.CHANNEL.send(
                    PacketDistributor.PLAYER.with(() -> player),
                    new ExplosionFlashPacket(flashIntensity, suppressionKick)
            );
        }
    }
